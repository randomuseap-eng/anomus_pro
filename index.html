  <!-- JAVASCRIPT LOGIC -->

  <script>

    // PERMANENT OWNER PASSCODE

    const PERMANENT_OWNER_CODE = "ACESS_OWNER_AP+ADMIN_UniQue";

    

    let isOwnerLoggedIn = false;

    let publishedProjects = [];

    let selectedPdfFile = null;


    // --- 1. SPLASH SCREEN ANIMATION (5 SECONDS) ---

    window.addEventListener('DOMContentLoaded', () => {

      const progressBar = document.getElementById('splash-progress');

      const percentText = document.getElementById('splash-percent');

      const statusText = document.getElementById('splash-status');

      const splashScreen = document.getElementById('splash-screen');


      let progress = 0;

      const interval = setInterval(() => {

        progress += 2;

        progressBar.style.width = `${progress}%`;

        percentText.innerText = `${progress}%`;


        if (progress === 30) statusText.innerText = "VERIFYING CORE SCHEMATICS...";

        if (progress === 70) statusText.innerText = "READY FOR DISPATCH...";


        if (progress >= 100) {

          clearInterval(interval);

          setTimeout(() => {

            splashScreen.style.opacity = '0';

            setTimeout(() => {

              splashScreen.style.display = 'none';

            }, 700);

          }, 300);

        }

      }, 90); // ~5 seconds duration

    });


    // --- 2. AUTHENTICATION & ACCESS CONTROL ---

    function openAuthModal() {

      if (isOwnerLoggedIn) {

        // Logout if already logged in

        isOwnerLoggedIn = false;

        updateRoleUI();

        renderProjects(publishedProjects);

        return;

      }

      document.getElementById('auth-modal').classList.remove('hidden');

      document.getElementById('passcode-input').value = '';

      document.getElementById('auth-error').classList.add('hidden');

    }


    function closeAuthModal() {

      document.getElementById('auth-modal').classList.add('hidden');

    }


    function verifyPasscode() {

      const input = document.getElementById('passcode-input').value.trim();

      if (input === PERMANENT_OWNER_CODE) {

        isOwnerLoggedIn = true;

        closeAuthModal();

        updateRoleUI();

        renderProjects(publishedProjects);

      } else {

        document.getElementById('auth-error').classList.remove('hidden');

      }

    }


    function updateRoleUI() {

      const roleBadge = document.getElementById('role-badge');

      const authBtn = document.getElementById('auth-btn');

      const publishBtn = document.getElementById('publish-btn');


      if (isOwnerLoggedIn) {

        roleBadge.className = "px-3 py-1 rounded-full text-xs font-mono bg-cyan-500/20 text-cyan-400 border border-cyan-500/40 flex items-center gap-1.5";

        roleBadge.innerHTML = '<i class="fa-solid fa-user-check text-cyan-400"></i><span>OWNER MODE</span>';

        

        authBtn.innerHTML = '<i class="fa-solid fa-right-from-bracket"></i><span>LOGOUT</span>';

        authBtn.className = "px-3 py-1.5 rounded-lg text-xs font-orbitron font-bold bg-slate-900 text-slate-400 hover:text-white border border-slate-800 transition-all";

        

        publishBtn.classList.remove('hidden');

      } else {

        roleBadge.className = "px-3 py-1 rounded-full text-xs font-mono bg-slate-800 text-slate-400 border border-slate-700 flex items-center gap-1.5";

        roleBadge.innerHTML = '<i class="fa-solid fa-user-shield text-slate-400"></i><span>VIEWER MODE</span>';

        

        authBtn.innerHTML = '<i class="fa-solid fa-lock"></i><span>OWNER LOGIN</span>';

        authBtn.className = "px-4 py-2 rounded-lg text-xs font-orbitron font-bold tracking-wider bg-slate-800 hover:bg-slate-700 text-cyan-400 border border-cyan-500/30 transition-all flex items-center gap-2";

        

        publishBtn.classList.add('hidden');

      }

    }


    // --- 3. PUBLISH MODAL & FILE HANDLING ---

    function openPublishModal() {

      if (!isOwnerLoggedIn) return;

      document.getElementById('publish-modal').classList.remove('hidden');

    }


    function closePublishModal() {

      document.getElementById('publish-modal').classList.add('hidden');

      document.getElementById('publish-form').reset();

      document.getElementById('file-label').innerText = "Click to choose PDF file";

      document.getElementById('file-error').classList.add('hidden');

      selectedPdfFile = null;

    }


    function handleFileSelect(e) {

      const file = e.target.files[0];

      const maxMb = 100;


      if (file) {

        if (file.size > maxMb * 1024 * 1024) {

          document.getElementById('file-error').classList.remove('hidden');

          e.target.value = '';

          selectedPdfFile = null;

          document.getElementById('file-label').innerText = "Click to choose PDF file";

        } else {

          document.getElementById('file-error').classList.add('hidden');

          selectedPdfFile = file;

          const sizeMb = (file.size / (1024 * 1024)).toFixed(2);

          document.getElementById('file-label').innerText = `${file.name} (${sizeMb} MB)`;

        }

      }

    }


    function handlePublish(e) {

      e.preventDefault();

      if (!selectedPdfFile) return;


      const title = document.getElementById('build-title').value;

      const category = document.getElementById('build-category').value;

      const description = document.getElementById('build-description').value;


      // Create browser Blob URL for session PDF access

      const pdfUrl = URL.createObjectURL(selectedPdfFile);


      const newProject = {

        id: Date.now(),

        title,

        category,

        description,

        pdfName: selectedPdfFile.name,

        pdfSize: (selectedPdfFile.size / (1024 * 1024)).toFixed(2),

        pdfUrl: pdfUrl,

        date: new Date().toLocaleDateString()

      };


      publishedProjects.unshift(newProject);

      renderProjects(publishedProjects);

      closePublishModal();

    }


    // --- 4. DELETE PROJECT (OWNER ONLY) ---

    function deleteProject(id) {

      if (!isOwnerLoggedIn) return;

      publishedProjects = publishedProjects.filter(p => p.id !== id);

      renderProjects(publishedProjects);

    }


    // --- 5. RENDER PROJECTS GRID & CUSTOMER ACCESS ---

    function renderProjects(projects) {

      const container = document.getElementById('projects-container');

      const emptyState = document.getElementById('empty-state');


      if (projects.length === 0) {

        container.innerHTML = '';

        container.appendChild(emptyState);

        emptyState.classList.remove('hidden');

        return;

      }


      container.innerHTML = '';

    // --- 6. PDF VIEWER MODAL FOR CUSTOMERS ---

    function openPdfModal(url, title, filename) {

      document.getElementById('pdf-modal').classList.remove('hidden');

      document.getElementById('pdf-modal-title').innerText = title.toUpperCase();

      document.getElementById('pdf-frame').src = url;

      const downloadBtn = document.getElementById('pdf-modal-download');

      downloadBtn.href = url;

      downloadBtn.download = filename;

    }


    function closePdfModal() {

      document.getElementById('pdf-modal').classList.add('hidden');

      document.getElementById('pdf-frame').src = '';

    }

      event.target.className = "cat-btn px-4 py-1.5 rounded-full text-xs font-mono bg-cyan-500/20 text-cyan-400 border border-cyan-500/50";


      if (cat === 'ALL') {

        renderProjects(publishedProjects);

      } else {

        const filtered = publishedProjects.filter(p => p.category === cat);

        renderProjects(filtered);

      }

    }
